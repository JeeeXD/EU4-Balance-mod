can_colonize_province = {
	condition = {
		tooltip = colonial_cannot_colonize
		potential = {
			FROM = {
				is_colonial_nation = yes
			}
		}
		allow = {
			FROM = {
				capital_scope = {
					ROOT = {
						colonial_region = PREV
					}
				}
			}
		}
	}
}

		potential = {
			always = yes
		}
		allow = {
			ai = no
		}
	}
}

can_make_core = {
	condition = {
		tooltip = ME_making_state_cores_is_toggled_off
		potential = {
			owner = { has_country_flag = ME_toggle_no_full_cores }
			is_core = owner
		}
		allow = {
			OR = {
				NOT = {
					owner = { has_country_flag = ME_toggle_no_full_cores }
				}
				NOT = { is_core = owner }
			}
		}
	}
}
