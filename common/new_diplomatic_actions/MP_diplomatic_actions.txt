create_influenced_free_city = {
	category = influence

	alert_index = 17 #doesn't matter. it is forced onto the subject
	alert_tooltip = sell_indulgence_alert_tooltip #doesn't matter. it is forced onto the subject

	require_acceptance = no

	is_visible = {
		ROOT ={
			is_part_of_hre = yes
		}
		FROM = {
			is_subject_of = ROOT
			is_part_of_hre = yes
			NOT = {
				num_of_cities = 2
			}
			NOT = {
				has_reform = influenced_free_city
			}
		}
	}
	is_allowed = {
		NOT = {
			calc_true_if = {
			all_country = {
				OR = {
					has_reform = free_city
					has_reform = influenced_free_city
				}
			}
			amount = 12
		    }
		}
		
	}
	on_accept = {
		FROM = {
		if = {
			limit = {
				government = republic
			}
			remove_government_reform = oligarchy_reform
			remove_government_reform = free_city
			remove_government_reform = noble_elite_reform
			remove_government_reform = plutocratic_reform
			remove_government_reform = civic_republicanism_government
			add_government_reform = influenced_free_city
		}
		else = {
			change_government = republic
			remove_government_reform = oligarchy_reform
			remove_government_reform = free_city
			remove_government_reform = noble_elite_reform
			remove_government_reform = plutocratic_reform
			remove_government_reform = civic_republicanism_government
			add_government_reform = influenced_free_city
		}
		swap_non_generic_missions = yes
		}
		
	}

	ai_will_do = {
		always = no
	}
}
