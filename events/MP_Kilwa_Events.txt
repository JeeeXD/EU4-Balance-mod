namespace = mp_zan

# What to do with Mutapa Great Project
country_event = {
	id = mp_zan.1
	title = mp_zan.1.t
	desc = mp_zan.1.d
	picture = CONQUEST_eventPicture
	
	is_triggered_only = yes
	
	option = {
		name = mp_zan.1.a
        1184 = {
            if = {
                limit = { 
                    has_great_project = {
                        type = khami_ruins
                        tier = 1
                    }
                }
                add_great_project_tier = {
                    type = khami_ruins
                    tier = -1
                }
                add_prestige = 50
                add_years_of_income = 1
                add_adm_power = 100
                add_dip_power = 100
                add_mil_power = 100
            }
            else = {
                add_prestige = 25
                add_years_of_income = 0.5
                add_adm_power = 50
                add_dip_power = 50
                add_mil_power = 50
            }
        }
		ai_chance = {
			factor = 1
		}
	}
    option = {
        name = mp_zan.1.b
        1184 = {
            if = {
                limit = { 
                    has_great_project = {
                        type = khami_ruins
                        tier = 1
                    }
                }
                add_treasury = -500
                add_great_project_tier = {
                    type = khami_ruins
                    tier = 1
                }
            }
            
        }
    }
}
